cmake_minimum_required(VERSION 3.16)
project(KinetraConsumer LANGUAGES CXX)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(KINETRA_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(KINETRA_BUILD_DOTIO OFF CACHE BOOL "" FORCE)
set(KINETRA_BUILD_RUNTIME ${CONSUMER_GRAPHICS} CACHE BOOL "" FORCE)
set(KINETRA_BUILD_LAUNCHER ${CONSUMER_GRAPHICS} CACHE BOOL "" FORCE)
add_subdirectory("${KINETRA_SOURCE}" engine)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE Kinetra::Simulation)
if(CONSUMER_GRAPHICS)
    target_link_libraries(consumer PRIVATE Kinetra::Launcher)
    target_compile_definitions(consumer PRIVATE CONSUMER_GRAPHICS=1)
    add_subdirectory("${DOTIO_SOURCE}" dotio)
endif()
